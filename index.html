const greeting = "Hello, World!";

function say(text, { times = 1, loud = false } = {}) {
  const message = loud ? text.toUpperCase() : text;
  for (let i = 0; i < times; i++) console.log(message);
}

say(greeting, { times: 3, loud: true });
